## 阶段 1 读需求、列约束（20 分钟）

把教材 10.2 的三句摘录逐句拆开，一句需求对一行，先只填前四列：

| 规则 | 来源（需求原句） | 范围类型 | 归属层 |
| --- | --- | --- | --- |
| 开始早于结束 | 摘录第 2 句 | 跨字段 | service |
| 人员存在 | 摘录第 1 句 | 跨表 | service |
| 哨位存在 | 摘录第 1 句 | 跨表 | service |
| 同一哨位时间不重叠 | 摘录第 3 句 | 跨记录 | service |

常见错误：

- 把「人员存在」写成 schema 的枚举校验——人员是库里的数据，不是固定选项。
- 把「时间顺序」写成数据库 `CHECK`——本章要训练的是 service 层可读、可测试的规则，且 SQLite 旧库加约束需要迁移。
- 漏掉「哨位存在」，只校验人员。

## 阶段 2 分类与定层（20 分钟）

判据只有三条：

1. 只看单个字段就能判断 → schema；
2. 要看多个字段或别的表、甚至别的记录 → service；
3. 资源本身找不到 → 404，不是业务规则。

本章四条规则全部落在 service。把「校验顺序」「错误码与信息」「用例」三列补齐，结果就是 README 3.1 的七列表。顺序的理由是「先便宜、后昂贵」：时间比较不查库，人员与哨位各一次点查，重叠是一次范围查询。

```python
from sqlalchemy.orm import Session

from app.modules.department.crud import get_by_code as get_dept_by_code
from app.modules.workstation import crud


def _ensure_dept_exists(db: Session, dept_code: str):
    if not get_dept_by_code(db, dept_code):
        raise HTTPException(status_code=400, detail="所属单位不存在")


def list_workstations(db, params):
    offset = (params.page - 1) * params.page_size
    items, total = crud.list_page(db, offset, params.page_size, params.dept_code)
    return {"items": items, "total": total}


def create_workstation(db, data):
    if crud.get_by_id(db, data.workstation_id):
        raise HTTPException(status_code=400, detail="哨位编号已存在")
    _ensure_dept_exists(db, data.dept_code)
    return crud.create(db, data)


def get_workstation(db, workstation_id: str):
    workstation = crud.get_by_id(db, workstation_id)
    if not workstation:
        raise HTTPException(status_code=404, detail="哨位不存在")
    return workstation


def update_workstation(db, workstation_id: str, data):
    workstation = crud.get_by_id(db, workstation_id)
    if not workstation:
        raise HTTPException(status_code=404, detail="哨位不存在")
    payload = data.model_dump(exclude_unset=True)

        _ensure_dept_exists(db, payload["dept_code"])
    return crud.update(db, workstation, payload)


def delete_workstation(db, workstation_id: str) -> None:
    workstation = crud.get_by_id(db, workstation_id)
```

```python
from sqlalchemy.orm import Session

from app.modules.duty_record.model import DutyRecord


def get_by_id(db: Session, record_id: int):
    return db.query(DutyRecord).filter(DutyRecord.id == record_id).first()


def list_page(
    db: Session,
    offset: int,
    limit: int,
    personnel_certificate_number: str | None = None,
    workstation_id: str | None = None,
):
    query = db.query(DutyRecord)
    if personnel_certificate_number:
        query = query.filter(
            DutyRecord.personnel_certificate_number == personnel_certificate_number
        )
    if workstation_id:
        query = query.filter(DutyRecord.workstation_id == workstation_id)
    total = query.count()
    items = query.order_by(DutyRecord.id).offset(offset).limit(limit).all()
    return items, total


def find_overlap(db: Session, workstation_id: str, start_time, end_time):
    """查同一哨位上、与给定时间段重叠的执勤记录。

    重叠条件：已有开始 < 新结束 且 已有结束 > 新开始。
    首尾相接（已有结束 == 新开始）不满足第二个条件，不算冲突。
    """
    return (
        db.query(DutyRecord)
        .filter(
            DutyRecord.workstation_id == workstation_id,
            DutyRecord.start_time < end_time,
            DutyRecord.end_time > start_time,
        )
        .all()
    )


def create(db: Session, data):
    record = DutyRecord(**data.model_dump())
    db.add(record)
    db.commit()
    db.refresh(record)
    return record


def delete(db: Session, record):
    db.delete(record)
    db.commit()
```

`app/modules/duty_record/service.py`（三条规则与顺序的核心）：

```python
from fastapi import HTTPException
from sqlalchemy.orm import Session

from app.modules.duty_record import crud
from app.modules.personnel.crud import (
    get_by_certificate_number as get_personnel_by_certificate_number,
)
from app.modules.workstation.crud import get_by_id as get_workstation_by_id


def _validate_duty_record(db: Session, data):
    """按规则清单的顺序校验；全部通过后才允许写入。

    顺序：时间顺序（不查库）→ 人员（一次查询）→ 哨位（一次查询）→ 重叠（范围查询）。
    任何一条失败都抛 400，控制器与 crud 不会执行到 commit。
    """
    # 规则一：开始时间必须早于结束时间（跨字段）
    if data.start_time >= data.end_time:
        raise HTTPException(status_code=400, detail="开始时间必须早于结束时间")

    # 规则二：人员和哨位必须存在（跨表）
    if not get_personnel_by_certificate_number(
        db, data.personnel_certificate_number
    ):
        raise HTTPException(status_code=400, detail="人员不存在")
    if not get_workstation_by_id(db, data.workstation_id):
        raise HTTPException(status_code=400, detail="哨位不存在")

    # 规则三：同一哨位的时间不能重叠（跨记录）
    conflicts = crud.find_overlap(
        db, data.workstation_id, data.start_time, data.end_time
    )
    if conflicts:
        raise HTTPException(status_code=400, detail="该哨位在该时段已有执勤安排")


def list_duty_records(db, params):
    offset = (params.page - 1) * params.page_size
    items, total = crud.list_page(
        db,
        offset,
        params.page_size,
        params.personnel_certificate_number,
        params.workstation_id,
    )
    return {"items": items, "total": total}


def create_duty_record(db, data):
    """先完成全部校验，再写入；校验过程中不 commit。"""
    _validate_duty_record(db, data)
    return crud.create(db, data)


def get_duty_record(db, record_id: int):
    record = crud.get_by_id(db, record_id)
    if not record:
        raise HTTPException(status_code=404, detail="执勤记录不存在")
    return record


def delete_duty_record(db, record_id: int) -> None:
    record = crud.get_by_id(db, record_id)
    if not record:
        raise HTTPException(status_code=404, detail="执勤记录不存在")
    crud.delete(db, record)
```
